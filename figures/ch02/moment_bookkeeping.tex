% The commanded wrench at the TCP when the impedance is defined about a
% displaced centre of compliance. The complete commanded force f acts at the
% TCP, the virtual centre sits at p_c, and the displacement r_c between them is
% what turns part of that force into the coupling contribution r_c x f. Only
% commanded quantities appear: the model-estimated external wrench is an
% implementation signal and is not part of this derivation.
% Colours follow the drawn palette: blue for the virtual displacement, red for
% a force, black for the moment. Every line is solid, and none is grey or white.
\begin{tikzpicture}[
    scale=1.15,
    font=\small,
    tool/.style={draw=black, thick},
    levc/.style={-{Latex[length=2mm]}, blue, very thick},
    frc/.style={-{Latex[length=2.2mm]}, red, very thick},
    mom/.style={-{Latex[length=2mm]}, black, thick},
    lbl/.style={font=\small, inner sep=2pt},
    sub/.style={font=\footnotesize, inner sep=2pt},
  ]

  \draw[black] (0.3,0) -- (5.7,0);
  \node[sub, anchor=west] at (4.75,-0.34) {Surface};

  \begin{scope}[rotate around={10:(1.50,0.55)}]
    \draw[tool] (1.50,0.55) rectangle (4.30,1.15);
  \end{scope}

  \fill (2.83,1.09) circle (1.7pt);
  \node[lbl, below] at (3.05,0.70) {$p_{\mathrm{TCP}}$};
  \fill (5.30,2.25) circle (1.7pt);
  \node[lbl, right] at (5.42,2.25) {$p_c$};

  % the virtual shift of the impedance reference point
  \draw[levc] (2.83,1.09) -- (5.30,2.25);
  \node[sub, anchor=south, blue] at (4.55,2.00) {$r_c$};

  % the complete commanded force, applied at the TCP
  \draw[frc] (2.83,2.30) -- (2.83,1.26);
  \node[lbl, left, red] at (2.72,2.10) {$f$};

  % the complete commanded moment
  \draw[mom] (3.05,2.10) arc (165:15:0.45);
  \node[sub, above] at (3.49,2.54) {$m$};

\end{tikzpicture}
